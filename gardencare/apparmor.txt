# AppArmor profile for the GardenCare Home Assistant App (N51).
#
# The Supervisor loads this file automatically because of its name, and enforces
# it around the container. It is the last line of the file-upload defence
# described in `docs/SECURITY.md` § 4: the media pipeline decides what an upload
# *is*, and this decides what the process could do if that judgement were ever
# wrong.
#
# ## The shape of the decisions here
#
# **Deny what is dangerous, allow what is ordinary.** A profile so tight that a
# routine Python import fails is a profile the next maintainer switches off, and
# an add-on running with `apparmor: false` is worse than one running under a
# profile that stops the attacks that actually matter. So the denials below are
# specific and permanent -- no mounting, no kernel modules, no raw sockets, no
# reading another process's memory -- while ordinary reads of the runtime are
# allowed.
#
# **The container runs as root** -- there is no `USER` in the Dockerfile and the
# entrypoint does not drop privileges -- which is what makes the capability list
# load-bearing rather than decorative: root's power *is* its capability set.
# `dac_override` and friends stay because the entrypoint creates `/data` on a
# volume whose ownership GardenCare does not control; `sys_admin`, `sys_module`,
# `net_admin` and `sys_ptrace` go, because nothing here has any business with the
# kernel, the network stack or another process.
#
# **`/data` is the only place this add-on writes.** That is not merely
# convention here -- everything else is read-only under this profile, so a bug
# that tried to write a photograph outside `/data` would be refused rather than
# silently succeed and be lost on the next update.
#
# ## Three broken releases, and what replaced the guessing
#
# Versions of this file shipped that stopped the add-on from starting, in this
# order:
#
#     sh: 0: cannot open /usr/local/bin/gardencare-start: Permission denied
#     /usr/local/bin/gardencare-start: 48: alembic: Permission denied
#     python: error while loading shared libraries: libpython3.12.so.1.0
#
# Execute without read on a script; then a virtual environment inside `/app`
# that was readable and not executable; then libpython, in a directory this file
# had not covered, under a version it had guessed wrong -- `python3.11`, in an
# image built `FROM python:3.12-slim`.
#
# One cause: this file *described* a base image's internals instead of deriving
# them. That is a description this repository cannot keep true across a base
# image bump, and it was being verified on a household's Raspberry Pi.
#
# What replaced it: the read surface is now the whole read-only system, and the
# security of this profile lives in its **denials**. `backend/tests/packaging`
# checks concrete paths -- the entry point, `alembic`, libpython, an extension
# module, the database, its lock files -- against these globs, and compiles the
# profile with AppArmor's own parser where one is installed. Replayed against
# the three versions above, that test reports exactly the failures each one
# shipped.
#
# **HG-03 still means installing it** and confirming GardenCare starts, serves
# Ingress, stores a photograph and survives a restart. If something is still
# refused, the Supervisor log names it, and that name belongs in an issue rather
# than in a hurried `apparmor: false`.

#include <tunables/global>

profile gardencare flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # --- capabilities ----------------------------------------------------------
  # Kept: the entrypoint creates and writes /data on a Supervisor-managed
  # volume, and does so as root over an ownership it did not choose.
  capability dac_override,
  capability dac_read_search,
  capability chown,
  capability fowner,
  capability fsetid,
  capability setgid,
  capability setuid,
  capability kill,

  # Refused, permanently. Each of these would be a step *out* of the container,
  # and GardenCare has no use for any of them: it reads sensors through the
  # Supervisor proxy over HTTP, and touches no hardware.
  deny capability sys_admin,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability sys_ptrace,
  deny capability sys_boot,
  deny capability net_admin,
  deny capability net_raw,
  deny capability mac_admin,
  deny capability mac_override,

  # --- namespaces and the kernel --------------------------------------------
  deny mount,
  deny umount,
  deny pivot_root,
  deny ptrace (trace read),
  deny @{PROC}/@{pid}/mem rwklx,
  deny @{PROC}/sys/** w,
  deny @{PROC}/kcore rwklx,
  deny /sys/** wl,
  deny /boot/** rwklx,

  # --- network ---------------------------------------------------------------
  # Outbound TCP to the Supervisor proxy and, when the household configures
  # them, to Pl@ntNet, Mistral and GBIF. UDP for DNS. Nothing raw: GardenCare
  # never inspects or forges a packet, so a process that tried to would be
  # doing something GardenCare does not do.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network netlink raw,
  deny network raw,
  deny network packet,

  # --- the runtime ----------------------------------------------------------
  #
  # **Read broadly, write nowhere, execute where programs live.** This section
  # was three separate enumerations of "which files does a Python distribution
  # need", and each one shipped a broken add-on:
  #
  #     sh: 0: cannot open /usr/local/bin/gardencare-start: Permission denied
  #     /usr/local/bin/gardencare-start: 48: alembic: Permission denied
  #     python: error while loading shared libraries: libpython3.12.so.1.0
  #
  # The last one named a version this file had guessed wrong (3.11, in an image
  # built `FROM python:3.12-slim`) in a directory it had not covered at all. The
  # lesson is not "add that path too". It is that an allowlist of a base image's
  # internals is a list this repository cannot keep true -- it changes with every
  # base image bump, and it is verified on the household's Raspberry Pi.
  #
  # So the read surface is now the whole read-only system, and **the security of
  # this profile lives in what it denies**: no write outside /data, no mount, no
  # kernel module, no raw socket, no other process's memory, no capability that
  # leaves the container. Those are the properties worth having, and none of them
  # depends on knowing where libpython lives.
  /bin/** rix,
  /usr/bin/** rix,
  /usr/local/bin/** rix,
  /sbin/** rix,
  /usr/sbin/** rix,

  /usr/** mr,
  /lib/** mr,
  /lib64/** mr,
  /etc/** r,
  /run/** r,

  # The image is the artefact; a running container has no business editing it.
  deny /usr/** w,
  deny /bin/** w,
  deny /sbin/** w,
  deny /lib/** w,
  deny /lib64/** w,
  deny /etc/** w,

  # --- the application ------------------------------------------------------
  # Read, never write: the code a container runs should not be something that
  # container can change.
  /app/ r,
  /app/** r,
  deny /app/** w,

  # The virtual environment lives *inside* /app -- `PATH="/app/.venv/bin:$PATH"`
  # in the Dockerfile -- so `alembic` and `uvicorn` are here, not in
  # /usr/local/bin. `m` because compiled extension modules (`.so`) are
  # memory-mapped and Python cannot import them otherwise.
  /app/.venv/bin/** rix,
  /app/.venv/** mr,

  # --- the household's data -------------------------------------------------
  # The SQLite database (with its WAL and shared-memory files, hence `k` for
  # locking) and the photographs. The one writable place.
  /data/ rw,
  /data/** rwk,

  /tmp/ rw,
  /tmp/** rwk,

  # --- process self-inspection ----------------------------------------------
  # The whole of its own /proc entry, read-only and owner-scoped. Python, uvicorn
  # and httpx each read a different corner of it (`maps`, `fd`, `mountinfo`,
  # `status`), and enumerating them one by one is how the next start fails for a
  # file nobody thought of. The dangerous parts -- `mem`, `kcore`, writes to
  # `sys` -- are denied above, and a deny always wins over an allow.
  owner @{PROC}/@{pid}/** r,
  @{PROC}/sys/kernel/random/boot_id r,
  @{PROC}/meminfo r,
  @{PROC}/stat r,
  @{PROC}/cpuinfo r,

  # Read-only: container runtimes report memory and CPU limits here, and writes
  # remain refused by the deny rule above.
  /sys/** r,

  /dev/urandom r,
  /dev/random r,
  /dev/null rw,
  /dev/tty rw,
}
